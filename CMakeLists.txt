cmake_minimum_required(VERSION 3.20)
project(Ocaain LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(core-tests tests/core_tests.cpp)
target_include_directories(core-tests PRIVATE native)
add_test(NAME core COMMAND core-tests)

if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "GoldSrc requires an x86/Win32 build, not x64.")
  endif()
  set(HLSDK_ROOT "" CACHE PATH "Local official Valve Half-Life SDK checkout")
  if(NOT HLSDK_ROOT)
    include(FetchContent)
    FetchContent_Declare(hlsdk GIT_REPOSITORY https://github.com/ValveSoftware/halflife.git
      GIT_TAG b1b5cf5892918535619b2937bb927e46cb097ba1)
    FetchContent_MakeAvailable(hlsdk)
    set(HLSDK_ROOT "${hlsdk_SOURCE_DIR}")
  endif()
  add_library(client SHARED native/proxy.cpp native/bridge.cpp native/client.def)
  target_include_directories(client PRIVATE native
    "${HLSDK_ROOT}/cl_dll" "${HLSDK_ROOT}/common" "${HLSDK_ROOT}/engine"
    "${HLSDK_ROOT}/public" "${HLSDK_ROOT}/pm_shared")
  target_link_libraries(client PRIVATE opengl32 user32)
  set_target_properties(client PROPERTIES PREFIX "" OUTPUT_NAME client)
  if(MSVC)
    set_property(TARGET client PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(client PRIVATE /W4 /permissive- /utf-8)
  elseif(MINGW)
    target_link_options(client PRIVATE -static-libgcc -static-libstdc++ -static)
    target_compile_options(client PRIVATE -Wall -Wextra -Wno-cast-function-type)
  endif()
endif()
